Q.1) Find element at position [r,c]:
nCr = rCc = n! / r!(n-r)!
3C2= (3*2)/(2*1)
4C3= (4*3*2)/(3*2*1) = 24/6 = 4

Formula : 
start from res = 1
ele = prev result * ((row+1-col)/col)
[4,3] = 6*(4+1-3)/3 = 12/3 = 4

Code for Entire Pascal Triangle:

class Solution {
public:
    vector<vector<int>> generate(int numRows) {
        numRows--;
        vector<vector<int>> arr;
        int res=1;
        for(int row=0;row<=numRows;row++){
            vector<int> temp;
            for(int col=0;col<=row;col++){
                if(col!=0){
                    res= (res*((row+1-col))/col);
                }else{
                    res=1;
                }
                temp.push_back(res);
            }
            arr.push_back(temp);
        }
        return arr;
    }
};
