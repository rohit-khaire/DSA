Q. Print Rth row present in Pascal's Triangle 

rowIndex = 3
[1,3,3,1]



class Solution {
public:
    vector<int> getRow(int rowIndex) {
        long long res =1;
        vector<int> arr;
        for(int col=0;col<=rowIndex;col++){
            if(col!=0){
                res = res * (rowIndex+1-col);
                res = res/col;
            }else{
                res =1;
            }
            arr.push_back(res);
        }
        return arr;
    }
};
